\section*{Problem 1: the complexity class NP (30 points)}
\begin{questions}
	\question[15] Assume $A \in$ NP and $B \in$ P. Show that $A \setminus B \in$ NP
	\begin{solutionbox}{\stretch{-1}} \\
	Let $M_A$ be a nondeterministic Turing machine that decides A in polynomial time,
	Let $M_B$ be a deterministic Turing machine that decides B in polynomial time,
	we construct a nondeterministic Turing machine M that decides $A \setminus B$ in polynomial time:
	
	
	Let $M="$On input $\langle w\rangle$:
	\begin{enumerate}
		\item Run $M_A$ on input $\langle w\rangle$
		\item If $M_A$ rejects, reject,otherwise run $M_B$ on w 
		\item If $M_B$ accepts, reject, if $M_B$ rejects, accept"
	\end{enumerate}

    \vspace{2em}

    \textbf{\large Correctness:} \\
	 As $A \setminus B$ means the set of strings that are in A but not in B,
	 the decider M runs $M_A$ to check if w is in A, if it is in A,
	 we need to check if w is in B, as if it is in B, then it cannot be in$A \setminus B$,
	 thus we accept only if w is in A but not in B, for time, since we know A is in NP,
	 we know that $M_A$ runs in polynomial time for a nondeterministic Turing machine, so running $M_A$ takes polynomial time for M as M is nondeterministic,
	 similarly $M_B$ takes polynomial time for a deterministic Turing machine,
	 so it takes polynomial time for a nondeterministic Turing machine, thus M runs in polynomial time and as it decides $A \setminus B$, $A \setminus B \in $ NP  
    
	\end{solutionbox}
	\pagebreak
	\question[15] Assume that P $=$ NP. Give a polynomial-time algorithm for finding a satisfying assignment for a boolean formula $\phi$, if one exists.
	\begin{solutionbox}{\stretch{-1}}\\
		Given P $=$ NP, there must exist a polynomial-time decider $M_{SAT}$ for SAT. We construct a polynomial-time algorithm for finding a satisfying assignment for a boolean formula $\phi$. \\ 
		The algorithm will be done in pseudo code:
		\lstset{style=mystyle}
		\begin{lstlisting}[language=python, escapeinside={(*}{*)}]
		def Search Sat((* $\phi$ *)):
		    Run (* $M_{SAT}$ *) on (* $\langle \phi \rangle$ *)
		    if (*$M_{SAT}$*) rejects, output 0.
		    Let vars be the set of all vars in (*$\phi$*)
		    Let (*$\phi' = \phi$*) 
		    Let (*$t$*) be a table of (var:bool)
		    for (*$i$*) in vars: 
		        Set each instance of var (*$i \in \phi'$*) to T. 
		        Run (*$M_{SAT}$*) on (*$\langle \phi' \rangle$*), 
		        if accept, add (*$i$*):T to table (*$t$*),
		        if reject, then for all instances of (*$i \in \phi'$*), assign (*$i$*):F to table (*$t$*). 
		    output (*$\phi'$*)
		\end{lstlisting} 

		\vspace{2em}


		\textbf{\large Correctness:} \\ 
		Since the algorithm starts by running $M_{SAT}$ on $\phi$, we are guaranteed that $\phi$ is satisfactory.
		Then we just periodically pick variable values until we find a satisfying permutation.
		
		\vspace{2em}

		\textbf{\large Time complexity analysis} \\  
		Given P $=$ NP, a polynomial-time decider must exist for SAT. The rest of the algorithm runs in $O(n^{1+c} + n^2 + n^c)$, $n = |\phi|$, and $c$ is some constant: \\
		$M_{SAT}$ runs in some polynomial-time $O(n^c)$. \\
		We run $M_{SAT}$ $n+1$ times, once before the loop, and $n$ times in the for loop, so $O(n^{c+1})$. \\
		Setting each variable $i \in \phi$ is $O(n)$ but we do it $2n$ times, so $O(2n^2) = O(n^2)$.
	\end{solutionbox}

\section*{Problem 2: Polynomial time reductions and NP-completeness}
\begin{questions}
    \question[20] Consider the language \[B=\{\langle G,k,v \rangle \mid G \text{ is a graph with a $k$-clique containing $v$}\}\] 
    
    \begin{parts}
	\part Show that $B$ is in NP. 
	\part Give a polynomial-time reduction from CLIQUE to $B$. 
	\part Is $B$ NP-complete?
\end{parts}
    \begin{solutionbox}{\stretch{-1}} \\ 

	    Recall that for a graph, we denote it with $G=(V,E)$,
	    where V is the set of vertices, let $M_C$ be a nondeterministic Turing machine that decides CLIQUE that runs in polynomial time, 
	    we construct a nondeterministic Turing machine $M$ that decides B. 

    
	Let $M="$On input $\langle G,k,v \rangle$: 
	\begin{enumerate}
		\item For each $w\in V$, we check if $w=v$
		\item If there exists a w, such that $w=v$, we run $M_C$ on $\langle G, k\rangle$ and goto step 3, if no such w exists, we reject.
		\item If $M_C$ accepts, accepts, otherwise reject"
	\end{enumerate}

	\vspace{2em}

	\textbf{\large Correctness:}\\ 
	Based on the definition of $B$, it is almost the same as CLIQUE except the added condition of containing $v$, 
	so, we start by checking if $v \in V$ and if so, we can just run the decider for CLIQUE, if it doesn't exist, 
	then $G$ does not contain $v$ and we reject.
	CLIQUE is in NP, so there exists a nondeterministic Turing machine that decides CLIQUE in polynomial time, 
	and as we have one for loop that goes through $V$, its also polynomial, so this machine runs in polynomial time and as such $B \in$ NP
    \end{solutionbox}
\end{questions}
